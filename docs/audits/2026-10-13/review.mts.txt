import assert from "node:assert/strict";
import { randomUUID } from "node:crypto";
import { eq, and } from "drizzle-orm";
import { db } from "@/db";
import { orders, cards, paymentAttempts, paymentAnomalies } from "@/db/schema";
import { seedAuto } from "@/server/order/test-helpers";
import { createOrder } from "@/server/order/create-order";
import { handlePaymentNotify, closeOrder } from "@/server/order/fulfill-order";
import { buildMockNotifyBody } from "@/server/payment/mock";
import { epusdtSign } from "@/lib/signature";

const getOrder = (no: string) => db.select().from(orders).where(eq(orders.orderNo, no)).get()!;
const getAnomalies = (no: string) => db.select().from(paymentAnomalies).where(eq(paymentAnomalies.orderNo, no)).all();
const observations: unknown[] = [];

async function setup(state: string) {
  const p = seedAuto([`card-${randomUUID()}`, `card-${randomUUID()}`]);
  const r = await createOrder({slug:p.slug,email:"audit@example.test",quantity:1});
  const o = getOrder(r.orderNo);
  const oldTrade = `old-${randomUUID()}`;
  db.insert(paymentAttempts).values({orderNo:o.orderNo,provider:"mock",tradeId:oldTrade,
    amountCents:o.totalCents,currency:o.currency,paymentUrl:"https://example.invalid/old",
    expectedActualAmount:"1.000000",expirationTime:new Date(Date.now()-60000),
    status:"expired",isCurrent:false}).run();
  if(state === "closed") assert.equal(closeOrder(o.orderNo).ok,true);
  if(state === "expired") {
    handlePaymentNotify("mock",buildMockNotifyBody({orderNo:o.orderNo,tradeId:o.tradeId!,amountCents:BigInt(o.totalCents),currency:o.currency,status:3}));
  }
  if(state === "completed") {
    handlePaymentNotify("mock",buildMockNotifyBody({orderNo:o.orderNo,tradeId:o.tradeId!,amountCents:BigInt(o.totalCents),currency:o.currency,status:2}));
  }
  assert.equal(getOrder(o.orderNo).status,state);
  return {o:getOrder(o.orderNo),oldTrade,p};
}
function signed(o: ReturnType<typeof getOrder>,tradeId:string,actual?:string) {
  const raw=buildMockNotifyBody({orderNo:o.orderNo,tradeId,amountCents:BigInt(o.totalCents),currency:o.currency,status:2});
  if(actual === undefined) return raw;
  const body=JSON.parse(raw); delete body.signature; body.actual_amount=actual;
  return JSON.stringify({...body,signature:epusdtSign(body,process.env.MOCK_PAYMENT_TOKEN!)});
}
for(const state of ["pending","expired","closed","completed"]) {
  const {o,oldTrade,p}=await setup(state);
  const before=db.select().from(cards).where(and(eq(cards.productId,p.id),eq(cards.status,"sold"))).all().length;
  const result=handlePaymentNotify("mock",signed(o,oldTrade));
  const rows=getAnomalies(o.orderNo);
  assert.equal(rows.length,state === "pending" ? 1 : 0);
  assert.equal(getOrder(o.orderNo).status,state);
  const after=db.select().from(cards).where(and(eq(cards.productId,p.id),eq(cards.status,"sold"))).all().length;
  assert.equal(before,after);
  observations.push({case:`historical-success/${state}`,anomalies:rows.length,state:getOrder(o.orderNo).status,soldDelta:after-before,message:result.message});
}
{
  const {o}=await setup("pending");
  db.update(orders).set({expectedActualAmount:"1.000000"}).where(eq(orders.orderNo,o.orderNo)).run();
  db.update(paymentAttempts).set({expectedActualAmount:"1.000000"}).where(eq(paymentAttempts.tradeId,o.tradeId!)).run();
  const result=handlePaymentNotify("mock",signed(o,o.tradeId!,"2.000000"));
  assert.equal(getAnomalies(o.orderNo).length,1);
  assert.equal(getOrder(o.orderNo).status,"pending");
  observations.push({case:"current-actual-amount-mismatch",anomalies:1,state:"pending",message:result.message});
}
{
  const {o,oldTrade}=await setup("pending");
  const result=handlePaymentNotify("mock",signed(o,oldTrade,"2.000000"));
  assert.equal(getAnomalies(o.orderNo).length,1);
  observations.push({case:"historical-actual-amount-mismatch",anomalies:1,message:result.message});
}
{
  const {o}=await setup("pending");
  const result=handlePaymentNotify("mock",signed(o,`unknown-${randomUUID()}`));
  assert.equal(getAnomalies(o.orderNo).length,0);
  observations.push({case:"unknown-trade",anomalies:0,message:result.message});
}
{
  const {o}=await setup("completed");
  const result=handlePaymentNotify("mock",signed(o,o.tradeId!));
  assert.equal(getAnomalies(o.orderNo).length,0);
  observations.push({case:"completed-current-duplicate",anomalies:0,message:result.message});
}
console.log(JSON.stringify({baseline:"fa30978",database:process.env.DATABASE_PATH,observations},null,2));
